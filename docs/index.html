libpt

A minimalist C library for systems programming and penetration testing.

Pentest Functions

Built-in support for common post-exploitation tasks, encoding, and networking.

Zero Dependencies

Self-contained library with minimal reliance on external system libraries.

Strict Naming

Consistent prefixing and clear function signatures for predictable development.

Compilation

To compile programs using libpt, link with the -lpt flag:
gcc main.c -lpt -o myapp

Networking Example

Simple TCP server and client interaction using libpt.

#include <pt/net.h>
#include <pt/types.h>
#include <stdio.h>

/* 
 * Standalone example demonstrating a basic TCP server using libpt.
 * To compile: gcc example_net.c -lpt -o example_net
 */
int main() {
    /* TCP Server listening on 127.0.0.1:8080 */
    int server_sock = ptnetbind("127.0.0.1", 8080);
    
    if (server_sock != -1) {
        printf("Server listening on 127.0.0.1:8080...\n");
        
        /* Accept an incoming connection from a client */
        int client_sock = ptnetacc(server_sock);
        if (client_sock != -1) {
            char buffer[1024] = {0};
            /* Receive data from the connected client */
            int bytes = ptnetrcv(client_sock, (u8*)buffer, sizeof(buffer));
            
            if (bytes > 0) {
                printf("Received: %s\n", buffer);
                /* Send a response back to the client */
                ptnetsnd(client_sock, (u8*)"Hello from libpt!", 17);
            }
            /* Close the client connection */
            ptnetcls(client_sock);
        }
        /* Close the server listening socket */
        ptnetcls(server_sock);
    } else {
        printf("Failed to bind server socket.\n");
        return 1;
    }
    
    return 0;
}

Cryptography Example

Demonstrating XOR encryption followed by Base64 encoding.

#include <pt/cry.h>
#include <pt/enc.h>
#include <pt/types.h>
#include <stdio.h>

/* 
 * Standalone example demonstrating XOR encryption and Base64 encoding.
 * To compile: gcc example_cry.c -lpt -o example_cry
 */
int main() {
    /* Data to be protected and the XOR key */
    u8 data[] = "Secret Message";
    u32 data_len = 14;
    u8 key[] = "key123";
    u32 key_len = 6;
    char b64_out[128] = {0};

    /* 
     * Perform in-place XOR encryption. 
     * This function will cycle the key over the data buffer.
     */
    ptxore(data, data_len, key, key_len);

    /* 
     * Encode the resulting encrypted (binary) data into a Base64 string 
     * for safe transport or storage in text formats.
     */
    ptb64e(data, data_len, b64_out);

    printf("Encrypted and Base64 encoded: %s\n", b64_out);

    return 0;
}

Process Example

Finding a process by name and retrieving its full command line.

#include <pt/proc.h>
#include <pt/types.h>
#include <stdio.h>

/* 
 * Standalone example demonstrating process search and command line retrieval.
 * To compile: gcc example_proc.c -lpt -o example_proc
 */
int main() {
    /* Search for a running process named "firefox" */
    int pid = ptprocfind("firefox");
    
    if (pid != -1) {
        char cmdline[512] = {0};
        /* 
         * Retrieve the command line arguments used to start the process.
         * This reads from the system's process information (e.g., /proc on Linux).
         */
        if (ptproccmd(pid, cmdline, sizeof(cmdline)) == 0) {
            printf("Found firefox (PID: %d)\nCommand line: %s\n", pid, cmdline);
        } else {
            printf("Found firefox (PID: %d) but failed to retrieve cmdline.\n", pid);
        }
    } else {
        printf("Process 'firefox' not found.\n");
    }

    return 0;
}

File IO Example

Mapping a file into the process memory space for efficient reading.

#include <pt/io.h>
#include <pt/types.h>
#include <stdio.h>

/* 
 * Standalone example demonstrating memory-mapped file IO.
 * To compile: gcc example_io.c -lpt -o example_io
 */
int main() {
    /* 
     * Map the entire content of "config.dat" into the virtual address space.
     * This avoids manual buffer management and multiple read() syscalls.
     */
    u32 file_size = 0;
    void *map = ptiomap("config.dat", &file_size);

    if (map != NULL) {
        /* 
         * The file content can now be accessed directly via memory pointers.
         * For example, checking the first byte of the file:
         */
        u8 first_byte = ((u8*)map)[0];
        printf("Successfully mapped 'config.dat' (%u bytes).\n", file_size);
        printf("First byte: 0x%02x\n", first_byte);
        
        /* 
         * Once finished, the memory region must be unmapped to release resources.
         * The size passed must match the size returned by ptiomap.
         */
        ptiounmap(map, file_size);
    } else {
        printf("Failed to map 'config.dat'. Please ensure the file exists.\n");
        return 1;
    }

    return 0;
}
